(forall ((v Vehicle) (r Roadway) (t Int))
  (=> (and (on_roadway v r t)
           (or (= (roadway_pos v r t) RP_ShoulderRight)
               (= (roadway_pos v r t) RP_ShoulderLeft)))
      (and (shoulder_use_necessary_and_safe v t)
           (or (= (shoulder_reason v t) SR_StopStandPark)
               (= (shoulder_reason v t) SR_AccelBeforeEnteringMainLane)
               (= (shoulder_reason v t) SR_DecelBeforeRightTurn)
               (= (shoulder_reason v t) SR_PassSlowingOrStoppedOrDisabledOrLeftTurning)
               (= (shoulder_reason v t) SR_AllowFasterVehiclePass)
               (= (shoulder_reason v t) SR_TrafficControlDevice)
               (= (shoulder_reason v t) SR_AvoidCollision)))))
