(forall ((v Vehicle) (r Roadway) (t Int))
  (=> (and (on_roadway v r t)
           (not (movement_can_be_made_safely v t)))
      (and
        (forall ((a Access))
          (=> (and (access_on_roadway a r)
                   (or (= (access_kind a) AK_PrivateRoad)
                       (= (access_kind a) AK_Driveway)))
              (not (turning_into v a t))))
        (forall ((t2 Int))
          (=> (and (<= t t2)
                   (not (movement_can_be_made_safely v t2)))
              (and (not (intends_turn v TD_Left t2))
                   (not (intends_turn v TD_Right t2))
                   (not (intends_turn v TD_UTurn t2))
                   (not (intends_lane_change v TD_Left t2))
                   (not (intends_lane_change v TD_Right t2))))))))
