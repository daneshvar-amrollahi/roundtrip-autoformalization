(forall ((v_left Vehicle) (v_right Vehicle) (r Roadway) (l_target Lane) (l_from_left Lane) (l_from_right Lane) (t Int))
  (=> (and (= (roadway_flow r) FK_OneWay)
           (>= (roadway_lane_count r) 3)
           (distinct v_left v_right)
           (= (lane_on_roadway l_target) r)
           (= (lane_on_roadway l_from_left) r)
           (= (lane_on_roadway l_from_right) r)
           (distinct l_target l_from_left)
           (distinct l_target l_from_right)
           (distinct l_from_left l_from_right)
           (on_roadway v_left r t)
           (on_roadway v_right r t)
           (on_lane v_left l_target t)
           (on_lane v_right l_target t)
           (lane_adjacent l_target l_from_left)
           (lane_adjacent l_target l_from_right)
           (lane_adjacent l_from_left l_from_right)
           (intends_lane_change v_left TD_Left t)
           (intends_lane_change v_right TD_Right t)
           (movement_can_be_made_safely v_left t)
           (movement_can_be_made_safely v_right t))
      (yield_right_of_way_to_pedestrian v_left t)))
