(forall ((v Vehicle) (r Roadway) (t Int))
  (=> (and (on_roadway v r t)
           (= (roadway_flow r) FK_TwoWay)
           (= (roadway_lane_count r) 3)
           (= (roadway_pos v r t) RP_CenterLane)
           (not (intends_turn v TD_Left t))
           (not (directed_by_traffic_control_device v t)))
      (and (exists ((v_passed Vehicle))
             (and (not (= v_passed v))
                  (passing v v_passed t)))
           (left_side_clear_for_passing v r t)
           (forall ((v_other Vehicle))
             (=> (and (not (= v_other v))
                      (on_roadway v_other r t))
                 (safe-distance v v_other t))))))
