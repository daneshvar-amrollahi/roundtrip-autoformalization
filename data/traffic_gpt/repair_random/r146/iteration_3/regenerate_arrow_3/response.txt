(forall ((v_sc Vehicle) (v_ev Vehicle) (t Int))
  (=> (and (= (kind v_sc) VK_Streetcar)
           (is_authorized_emergency_vehicle v_ev)
           (emergency_vehicle_approaching v_sc v_ev t))
      (and (stopped v_sc t)
           (forall ((t2 Int))
             (=> (and (<= t t2)
                      (not (completely_passed v_ev v_sc t2)))
                 (and (stopped v_sc t2)
                      (forall ((i Intersection))
                        (not (in_intersection v_sc i t2)))))))))
