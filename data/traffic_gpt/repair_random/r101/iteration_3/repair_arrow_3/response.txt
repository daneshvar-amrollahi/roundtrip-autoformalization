(forall ((v Vehicle) (i Intersection) (r Roadway) (t Int))
  (=> (and (on_roadway v r t)
           (intersection_on_roadway i r)
           (approaching_intersection v i t)
           (< (dist_to_feature v PF_Crosswalk t) ft_20)
           (or (= (stop_action v t) SA_Stand)
               (= (stop_action v t) SA_Park))
           (not (and (standing_vehicle v t)
                     (is-stopping v t))))
      (and (not (= (stop_action v t) SA_Stand))
           (not (= (stop_action v t) SA_Park)))))
