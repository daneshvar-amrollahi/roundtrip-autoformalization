(forall ((v_right Vehicle) (v_left Vehicle) (l_target Lane) (l_from_right Lane) (l_from_left Lane) (r Roadway) (t Int))
  (=> (and (not (= v_right v_left))
           (= (lane_on_roadway l_target) r)
           (= (lane_on_roadway l_from_right) r)
           (= (lane_on_roadway l_from_left) r)
           (= (roadway_flow r) FK_OneWay)
           (>= (roadway_lane_count r) 3)
           (lane_adjacent l_from_right l_target)
           (lane_adjacent l_from_left l_target)
           (on_lane v_right l_from_right t)
           (on_lane v_left l_from_left t)
           (intends_lane_change v_right TD_Left t)
           (intends_lane_change v_left TD_Right t)
           (movement_can_be_made_safely v_right t)
           (movement_can_be_made_safely v_left t)
           (sufficient_space_for_vehicle_to_enter v_right v_left t)
           (sufficient_space_for_vehicle_to_enter v_left v_right t))
      (yield_right_of_way_to_pedestrian v_right t)))
