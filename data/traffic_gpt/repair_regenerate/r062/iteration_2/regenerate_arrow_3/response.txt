(forall ((v Vehicle) (t Int))
  (=> (and (opportunity_to_signal v t)
           (or (intends_turn v TD_Left t)
               (intends_turn v TD_Right t)
               (intends_turn v TD_UTurn t)
               (intends_lane_change v TD_Left t)
               (intends_lane_change v TD_Right t)
               (and (parked v t) (is-accelerating v t))))
      (and
        (=> (intends_turn v TD_Left t) (turn_signal_on v TD_Left t))
        (=> (intends_turn v TD_Right t) (turn_signal_on v TD_Right t))
        (=> (intends_turn v TD_UTurn t) (turn_signal_on v TD_Left t))
        (=> (intends_lane_change v TD_Left t) (turn_signal_on v TD_Left t))
        (=> (intends_lane_change v TD_Right t) (turn_signal_on v TD_Right t))
        (=> (and (parked v t) (is-accelerating v t))
            (or (turn_signal_on v TD_Left t) (turn_signal_on v TD_Right t))))))
