(forall ((v_op Vehicle) (v_em Vehicle) (r Roadway) (t Int))
  (=> (and (on_roadway v_op r t)
           (on_roadway v_em r t)
           (is_authorized_emergency_vehicle v_em)
           (stationary_with_signals v_em t)
           (emergency_vehicle_approaching v_op v_em t)
           (= (roadway_kind r) RK_Highway)
           (>= (roadway_lane_count r) 2)
           (= (roadway_flow r) FK_OneWay))
      (vacated_nearest_lane v_op v_em t)))
