(forall ((v Vehicle) (z Zone) (r Roadway) (t Int))
  (=> (and (is_safety_zone z)
           (on_roadway v r t)
           (in_zone v z t)
           (or (= (stop_action v t) SA_Stop)
               (= (stop_action v t) SA_Stand)
               (= (stop_action v t) SA_Park))
           (or (driving_through_safety_zone v z t)
               (<= (dist_to_feature v PF_SafetyZone t) ft_30)))
      (not (or (parked_on_roadway v r t)
               (standing_vehicle v t)
               (stopped v t)))))
