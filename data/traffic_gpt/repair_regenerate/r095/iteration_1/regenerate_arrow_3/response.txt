(forall ((v Vehicle) (r Roadway) (z Zone) (t Int))
  (=> (and (on_roadway v r t)
           (in_zone v z t)
           (is_safety_zone z)
           (or (driving_through_safety_zone v z t)
               (<= (distance v v t) (* ft_30 ft_30)))
           (or (= (stop_action v t) SA_Stop)
               (= (stop_action v t) SA_Stand)
               (= (stop_action v t) SA_Park)))
      (and (not (parked_on_roadway v r t))
           (not (standing_vehicle v t))
           (not (stopped v t)))))
