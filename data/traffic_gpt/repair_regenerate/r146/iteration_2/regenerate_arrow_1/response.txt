(forall ((v_sc Vehicle) (v_em Vehicle) (i Intersection) (t Int))
  (=> (and (= (kind v_sc) VK_Streetcar)
           (is_authorized_emergency_vehicle v_em)
           (emergency_vehicle_approaching v_sc v_em t))
      (and (stopped v_sc t)
           (not (in_intersection v_sc i t))
           (forall ((t2 Int))
             (=> (and (<= t t2)
                      (not (completely_passed v_em v_sc t2)))
                 (and (stopped v_sc t2)
                      (not (in_intersection v_sc i t2))))))))
