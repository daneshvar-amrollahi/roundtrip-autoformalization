(forall ((v_emerge Vehicle) (a Access) (r Roadway) (t Int) (v_app Vehicle))
  (=> (and (emerging_from v_emerge a t)
           (or (= (access_kind a) AK_Alley)
               (= (access_kind a) AK_Driveway)
               (= (access_kind a) AK_Building))
           (access_on_roadway a r)
           (on_roadway v_emerge r t)
           (approaching_opposite v_app v_emerge t)
           (not (= v_app v_emerge)))
      (yield_right_of_way_to_pedestrian v_emerge t)))
