(forall ((v Vehicle) (t Int))
  (=> (and (or (exists ((a Access))
                 (and (turning_into v a t)
                      (or (= (access_kind a) AK_PrivateRoad)
                          (= (access_kind a) AK_Driveway))))
             (exists ((d TurnDir))
               (intends_lane_change v d t))
             (exists ((d TurnDir))
               (executing_turn v d t)))
           (not (movement_can_be_made_safely v t)))
      (and (forall ((a Access))
             (=> (or (= (access_kind a) AK_PrivateRoad)
                     (= (access_kind a) AK_Driveway))
                 (not (turning_into v a t))))
           (forall ((d TurnDir))
             (not (intends_lane_change v d t)))
           (forall ((d TurnDir))
             (not (executing_turn v d t))))))
