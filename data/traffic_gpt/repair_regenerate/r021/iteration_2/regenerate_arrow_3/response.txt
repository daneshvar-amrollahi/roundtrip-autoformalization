(forall ((v_pass Vehicle) (v_passed Vehicle) (r Roadway) (t Int))
  (=> (and (passing v_pass v_passed t)
           (= (passing_side v_pass v_passed t) PS_Right)
           (same_roadway_as_bus v_pass v_passed r t))
      (and (safely_clear v_pass v_passed t)
           (or (intends_turn v_passed TD_Left t)
               (executing_turn v_passed TD_Left t))
           (or (and (roadway_has_marked_lanes r)
                    (or (= (roadway_pos v_pass r t) RP_RightLane)
                        (= (roadway_pos v_pass r t) RP_ShoulderRight)))
               (and (not (roadway_has_marked_lanes r))
                    (roadway_sufficient_width r)
                    (or (= (roadway_pos v_pass r t) RP_RightHalf)
                        (= (roadway_pos v_pass r t) RP_ShoulderRight)))))))
