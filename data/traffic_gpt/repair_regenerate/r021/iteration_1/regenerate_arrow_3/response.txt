(forall ((v_pass Vehicle) (v_other Vehicle) (r Roadway) (t Int))
  (=> (and (not (= v_pass v_other))
           (passing v_pass v_other t)
           (= (passing_side v_pass v_other t) PS_Right)
           (same_roadway_as_bus v_pass v_other r t))
      (and (movement_can_be_made_safely v_pass t)
           (or (intends_turn v_other TD_Left t)
               (executing_turn v_other TD_Left t))
           (or (and (roadway_has_marked_lanes r)
                    (>= (roadway_lane_count r) 2))
               (and (not (roadway_has_marked_lanes r))
                    (roadway_sufficient_width r))
               (and (= (roadway_pos v_other r t) RP_LeftHalf)
                    (roadway_free_of_obstructions r t))))))
