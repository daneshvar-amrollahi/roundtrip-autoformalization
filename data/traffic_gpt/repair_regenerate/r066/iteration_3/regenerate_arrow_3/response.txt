(forall ((v_term Vehicle) (r_term Roadway) (i Intersection) (t Int))
  (=> (and (on_roadway v_term r_term t)
           (road_terminates_at r_term i)
           (= (intersection_control i) IC_Uncontrolled)
           (or (= (roadway_lane_count r_term) 1)
               (= (roadway_lane_count r_term) 2))
           (or (= (roadway_divided r_term) DK_Divided)
               (and (roadway_has_marked_lanes r_term)
                    (>= (roadway_lane_count r_term) 3))))
      (forall ((v_hazard Vehicle) (r_other Roadway))
        (=> (and (not (= v_hazard v_term))
                 (not (= r_other r_term))
                 (on_roadway v_hazard r_other t)
                 (or (in_intersection v_hazard i t)
                     (approaching_intersection v_hazard i t))
                 (is_immediate_hazard v_hazard v_term i t))
            (and (stopped v_term t)
                 (yield_right_of_way_to_pedestrian v_term t)
                 (not (entering_intersection v_term i t)))))))
