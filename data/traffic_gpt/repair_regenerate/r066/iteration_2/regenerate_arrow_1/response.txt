(forall ((v_op Vehicle) (r_op Roadway) (i Intersection) (t Int))
  (=> (and (on_roadway v_op r_op t)
           (approaching_intersection v_op i t)
           (= (intersection_control i) IC_Uncontrolled)
           (intersection_on_roadway i r_op)
           (or (= (roadway_lane_count r_op) 1)
               (= (roadway_lane_count r_op) 2))
           (or (= (roadway_divided r_op) DK_Divided)
               (and (roadway_has_marked_lanes r_op)
                    (>= (roadway_lane_count r_op) 3))))
      (forall ((v_other Vehicle) (r_other Roadway))
        (=> (and (not (= v_other v_op))
                 (intersection_on_roadway i r_other)
                 (not (= r_other r_op))
                 (on_roadway v_other r_other t)
                 (or (in_intersection v_other i t)
                     (approaching_intersection v_other i t))
                 (is_immediate_hazard v_other v_op i t))
            (and (stopped v_op t)
                 (yield_right_of_way_to_pedestrian v_op t)
                 (not (entering_intersection v_op i t)))))))
