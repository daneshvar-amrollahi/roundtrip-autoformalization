(forall ((v_follower Vehicle) (v_fire Vehicle) (t Int))
  (=> (and (not (on_official_business v_follower t))
           (following_fire_apparatus v_follower v_fire t)
           (= (kind v_fire) VK_FireApparatus)
           (emergency_signals_active v_fire t))
      (not (< (distance v_follower v_fire t) ft_500_sq))))
