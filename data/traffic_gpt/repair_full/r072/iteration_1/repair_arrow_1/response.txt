(forall ((v_op Vehicle) (i Intersection) (t Int))
  (=> (and (approaching_intersection v_op i t)
           (= (intersection_control i) IC_YieldSign))
      (and (reduced_speed_appropriate v_op t)
           (forall ((v_other Vehicle))
             (=> (and (not (= v_other v_op))
                      (or (in_intersection v_other i t)
                          (and (exists ((r_other Roadway) (r_op Roadway))
                                 (and (on_roadway v_other r_other t)
                                      (on_roadway v_op r_op t)
                                      (intersection_on_roadway i r_other)
                                      (intersection_on_roadway i r_op)
                                      (= (roadway_kind r_other) RK_Highway)
                                      (= (roadway_kind r_op) RK_Highway)
                                      (not (= r_other r_op))))
                               (approaching_intersection v_other i t)
                               (is_immediate_hazard v_other v_op i t))))
                 (not (entering_intersection v_op i t))))))))
