(forall ((v Vehicle) (r Roadway) (v_oncoming Vehicle) (t Int))
  (=> (and (on_roadway v r t)
           (obstruction_present v r t)
           (crossing_center_line v r t)
           (not (= v v_oncoming))
           (on_roadway v_oncoming r t)
           (approaching_opposite v_oncoming v t)
           (roadway_free_of_obstructions r t))
      (yield_right_of_way_to_pedestrian v t)))
