(forall ((v Vehicle) (ev Vehicle) (r Roadway) (t Int))
  (=> (and (on_roadway v r t)
           (on_roadway ev r t)
           (is_authorized_emergency_vehicle ev)
           (emergency_vehicle_approaching ev v t))
      (exists ((t_pull Int))
        (and (<= t t_pull)
             (standing_vehicle v t_pull)
             (parallel_to_curb v r t_pull)
             (close_to_right_curb v r t_pull)
             (forall ((i Intersection))
               (not (in_intersection v i t_pull)))
             (forall ((t2 Int))
               (=> (and (<= t_pull t2)
                        (not (completely_passed ev v t2)))
                   (and (standing_vehicle v t2)
                        (forall ((i Intersection))
                          (not (in_intersection v i t2))))))))))
