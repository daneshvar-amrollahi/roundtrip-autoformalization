(forall ((v_op Vehicle) (v_em Vehicle) (r Roadway) (t Int))
  (=> (and (on_roadway v_op r t)
           (on_roadway v_em r t)
           (distinct v_op v_em)
           (is_authorized_emergency_vehicle v_em)
           (emergency_vehicle_approaching v_em v_op t))
      (exists ((t_pull Int))
        (and (<= t t_pull)
             (parallel_to_curb v_op r t_pull)
             (close_to_right_curb v_op r t_pull)
             (forall ((i Intersection))
               (not (in_intersection v_op i t_pull)))
             (standing_vehicle v_op t_pull)
             (forall ((t2 Int))
               (=> (and (<= t_pull t2)
                        (not (completely_passed v_em v_op t2)))
                   (and (standing_vehicle v_op t2)
                        (forall ((i2 Intersection))
                          (not (in_intersection v_op i2 t2))))))))))
