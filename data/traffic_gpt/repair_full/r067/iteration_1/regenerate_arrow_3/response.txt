(forall ((v_unpaved Vehicle) (v_paved Vehicle) (i Intersection) (r_unpaved Roadway) (r_paved Roadway) (t Int))
  (=> (and (on_roadway v_unpaved r_unpaved t)
           (not (roadway_is_paved r_unpaved))
           (approaching_intersection v_unpaved i t)
           (intersection_on_roadway i r_paved)
           (roadway_is_paved r_paved)
           (on_roadway v_paved r_paved t)
           (not (= v_unpaved v_paved))
           (or (in_intersection v_paved i t)
               (approaching_intersection v_paved i t))
           (is_immediate_hazard v_paved v_unpaved i t))
      (and (stopped v_unpaved t)
           (forall ((t2 Int))
             (=> (and (<= t t2)
                      (or (in_intersection v_paved i t2)
                          (entering_intersection v_paved i t2)))
                 (not (entering_intersection v_unpaved i t2)))))))
