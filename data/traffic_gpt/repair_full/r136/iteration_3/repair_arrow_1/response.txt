(forall ((v Vehicle) (t Int))
  (=> (and (on_sidewalk_or_trail v t)
           (not (and (exists ((r Roadway))
                       (and (on_roadway v r t)
                            (= (roadway_pos v r t) RP_Sidewalk)))
                     (exists ((a Access) (r2 Roadway))
                       (and (access_on_roadway a r2)
                            (access_kind a AK_Driveway)
                            (on_roadway v r2 t))))))
      (not (is-moving v t))))
