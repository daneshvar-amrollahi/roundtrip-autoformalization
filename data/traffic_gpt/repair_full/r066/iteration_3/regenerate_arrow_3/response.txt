(forall ((v Vehicle) (i Intersection) (r_app Roadway) (t Int))
  (=> (and (on_roadway v r_app t)
           (or (= (roadway_lane_count r_app) 1)
               (= (roadway_lane_count r_app) 2))
           (approaching_intersection v i t)
           (intersection_on_roadway i r_app)
           (= (intersection_control i) IC_Uncontrolled)
           (exists ((r_int Roadway))
             (and (intersection_on_roadway i r_int)
                  (or (= (roadway_divided r_int) DK_Divided)
                      (and (roadway_has_marked_lanes r_int)
                           (>= (roadway_lane_count r_int) 3))))))
      (and (stopped v t)
           (forall ((v_other Vehicle) (r_other Roadway))
             (=> (and (intersection_on_roadway i r_other)
                      (not (= r_other r_app))
                      (on_roadway v_other r_other t)
                      (or (in_intersection v_other i t)
                          (approaching_intersection v_other i t))
                      (is_immediate_hazard v_other v i t))
                 (not (entering_intersection v i t)))))))
