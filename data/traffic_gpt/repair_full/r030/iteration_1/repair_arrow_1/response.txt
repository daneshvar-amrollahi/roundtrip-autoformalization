(forall ((v_right Vehicle) (v_left Vehicle) (l_target Lane) (l_from_right Lane) (l_from_left Lane) (t Int))
  (=> (and (not (= v_right v_left))
           (on_roadway v_right (lane_on_roadway l_target) t)
           (on_roadway v_left (lane_on_roadway l_target) t)
           (= (roadway_flow (lane_on_roadway l_target)) FK_OneWay)
           (>= (roadway_lane_count (lane_on_roadway l_target)) 3)
           (lane_adjacent l_from_right l_target)
           (lane_adjacent l_from_left l_target)
           (on_lane v_right l_from_right t)
           (on_lane v_left l_from_left t)
           (intends_lane_change v_right TD_Left t)
           (intends_lane_change v_left TD_Right t)
           (on_lane v_right l_target (+ t 1))
           (on_lane v_left l_target (+ t 1)))
      (yield_right_of_way_to_pedestrian v_right t)))
