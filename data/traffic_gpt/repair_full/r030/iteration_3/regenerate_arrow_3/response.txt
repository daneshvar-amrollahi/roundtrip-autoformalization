(forall ((r Roadway) (v_right Vehicle) (v_other Vehicle) (l_right Lane) (l_other Lane) (l_target Lane) (t Int))
  (=> (and (= (roadway_flow r) FK_OneWay)
           (>= (roadway_lane_count r) 3)
           (= (lane_on_roadway l_right) r)
           (= (lane_on_roadway l_other) r)
           (= (lane_on_roadway l_target) r)
           (not (= v_right v_other))
           (not (= l_right l_other))
           (lane_adjacent l_right l_target)
           (lane_adjacent l_other l_target)
           (on_lane v_right l_right t)
           (on_lane v_other l_other t)
           (on_lane v_right l_target (+ t 1))
           (on_lane v_other l_target (+ t 1))
           (exists ((i Intersection))
             (and (approaching_intersection v_right i t)
                  (pedestrian_present_in_crosswalk i t))))
      (yield_right_of_way_to_pedestrian v_right t)))
