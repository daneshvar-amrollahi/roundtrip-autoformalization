(forall ((v Vehicle) (a Access) (r Roadway) (t Int))
  (=> (and (emerging_from v a t)
           (access_on_roadway a r)
           (or (= (access_kind a) AK_Alley)
               (= (access_kind a) AK_Driveway)
               (= (access_kind a) AK_Building))
           (on_roadway v r t)
           (or (= (district_of_roadway r) DK_Business)
               (= (district_of_roadway r) DK_Residence))
           (crossing_sidewalk v t))
      (stopped_before_sidewalk v t)))
