(forall ((v Vehicle) (sc Vehicle) (r Roadway) (t Int))
  (=> (and (not (= v sc))
           (on_roadway v r t)
           (on_roadway sc r t)
           (= (roadway_pos v r t) RP_StreetcarTrack)
           (= (roadway_pos sc r t) RP_StreetcarTrack)
           (in_front_of v sc t)
           (= (kind sc) VK_Streetcar)
           (streetcar_operator_signaled sc v t))
      (moved_off_track_asap v t)))
