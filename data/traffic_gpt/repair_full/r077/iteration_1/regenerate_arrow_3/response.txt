(forall ((v_op Vehicle) (v_em Vehicle) (r Roadway) (l Lane) (t Int))
  (=> (and (not (= v_op v_em))
           (on_roadway v_op r t)
           (on_roadway v_em r t)
           (= (roadway_kind r) RK_Highway)
           (>= (roadway_lane_count r) 2)
           (is_authorized_emergency_vehicle v_em)
           (approaching_bus v_op v_em t)
           (stationary_with_signals v_em t)
           (emergency_signals_active v_em t)
           (on_lane v_em l t)
           (lane_on_roadway l)
           (= (lane_on_roadway l) r))
      (vacated_nearest_lane v_op v_em t)))
