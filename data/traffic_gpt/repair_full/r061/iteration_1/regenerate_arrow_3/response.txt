(forall ((v Vehicle) (a Access) (t Int))
  (=> (and (movement_can_be_made_safely v t)
           (or (intends_turn v TD_Left t)
               (intends_turn v TD_Right t)
               (intends_turn v TD_UTurn t)
               (intends_lane_change v TD_Left t)
               (intends_lane_change v TD_Right t)
               (turning_into v a t)))
      (and (not (executing_turn v TD_Left t))
           (not (executing_turn v TD_Right t))
           (not (executing_turn v TD_UTurn t))
           (not (intends_lane_change v TD_Left t))
           (not (intends_lane_change v TD_Right t))
           (not (turning_into v a t)))))
