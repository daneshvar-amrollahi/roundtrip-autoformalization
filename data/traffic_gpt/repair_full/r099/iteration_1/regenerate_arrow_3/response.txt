(forall ((v Vehicle) (t Int))
  (=> (and (in_front_of_driveway v t)
           (or (= (stop_action v t) SA_Stand)
               (= (stop_action v t) SA_Park)))
      (and (forall ((t_prev Int))
             (=> (< t_prev t)
                 (in_front_of_driveway v t_prev)))
           (forall ((t_next Int))
             (=> (> t_next t)
                 (in_front_of_driveway v t_next))))))
