(forall ((v Vehicle) (ev Vehicle) (r Roadway) (t Int))
  (=> (and (on_roadway v r t)
           (on_roadway ev r t)
           (is_authorized_emergency_vehicle ev)
           (stationary_with_signals ev t)
           (not (vacated_nearest_lane v ev t)))
      (and (reduced_speed_appropriate v t)
           (=> (and (speed_limit_posted r t)
                    (< (speed_limit_value r t) mph_25))
               (not (> (velocity v t) mph_5)))
           (=> (and (speed_limit_posted r t)
                    (not (< (speed_limit_value r t) mph_25)))
               (not (> (velocity v t) (- (speed_limit_value r t) mph_20)))))))
