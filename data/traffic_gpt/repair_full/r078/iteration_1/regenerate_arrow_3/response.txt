(forall ((v_op Vehicle) (v_em Vehicle) (r Roadway) (t Int))
  (=> (and (on_roadway v_op r t)
           (on_roadway v_em r t)
           (speed_limit_posted r t)
           (emergency_vehicle_approaching v_op v_em t)
           (stationary_with_signals v_em t))
      (forall ((t2 Int))
        (=> (and (<= t t2)
                 (not (vacated_nearest_lane v_op v_em t2)))
            (and (=> (< (speed_limit_value r t2) mph_25)
                     (<= (velocity v_op t2) (- (speed_limit_value r t2) mph_5)))
                 (=> (>= (speed_limit_value r t2) mph_25)
                     (<= (velocity v_op t2) (- (speed_limit_value r t2) mph_20))))))))
