(forall ((v Vehicle) (ev Vehicle) (r Roadway) (t Int))
  (=> (and (emergency_vehicle_approaching v ev t)
           (stationary_with_signals ev t)
           (on_roadway v r t)
           (on_roadway ev r t)
           (not (vacated_nearest_lane v ev t))
           (speed_limit_posted r t))
      (<= (velocity v t)
          (- (speed_limit_value r t)
             (ite (< (speed_limit_value r t) mph_25) mph_5 mph_20)))))
