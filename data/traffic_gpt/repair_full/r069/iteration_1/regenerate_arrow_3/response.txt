(forall ((v_term Vehicle) (i Intersection) (r_term Roadway) (t Int))
  (=> (and (approaching_intersection v_term i t)
           (on_roadway v_term r_term t)
           (road_terminates_at r_term i))
      (forall ((v_other Vehicle) (r_other Roadway))
        (=> (and (not (= v_other v_term))
                 (on_roadway v_other r_other t)
                 (not (= r_other r_term))
                 (or (in_intersection v_other i t)
                     (approaching_intersection v_other i t))
                 (is_immediate_hazard v_other v_term i t))
            (and (stopped v_term t)
                 (forall ((t2 Int))
                   (=> (and (<= t t2)
                            (or (in_intersection v_other i t2)
                                (approaching_intersection v_other i t2))
                            (is_immediate_hazard v_other v_term i t2))
                       (and (stopped v_term t2)
                            (not (entering_intersection v_term i t2))))))))))
