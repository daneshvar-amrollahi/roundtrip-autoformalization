(forall ((v Vehicle) (i Intersection) (r_term Roadway) (t Int))
  (=> (and (approaching_intersection v i t)
           (on_roadway v r_term t)
           (road_terminates_at r_term i))
      (and (stopped v t)
           (forall ((v_other Vehicle) (r_other Roadway))
             (=> (and (not (= v_other v))
                      (intersection_on_roadway i r_other)
                      (not (= r_other r_term))
                      (on_roadway v_other r_other t)
                      (or (in_intersection v_other i t)
                          (approaching_intersection v_other i t))
                      (is_immediate_hazard v_other v i t))
                 (and (not (entering_intersection v i t))
                      (not (in_intersection v i t))))))))
