(forall ((v_follower Vehicle) (v_leader Vehicle) (v_other Vehicle) (r Roadway) (t Int))
  (=> (and (following v_follower v_leader t)
           (same_roadway_as_bus v_follower v_leader r t)
           (on_roadway v_follower r t)
           (on_roadway v_leader r t)
           (not (= (district_of_roadway r) DK_Business))
           (not (= (district_of_roadway r) DK_Residence))
           (or (= (kind v_follower) VK_Truck) (drawing_another_vehicle v_follower t))
           (or (= (kind v_leader) VK_Truck) (drawing_another_vehicle v_leader t))
           (movement_can_be_made_safely v_follower t)
           (not (= v_other v_follower))
           (not (= v_other v_leader)))
      (sufficient_space_for_vehicle_to_enter v_other v_follower t)))
