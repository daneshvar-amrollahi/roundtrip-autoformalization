(forall ((v_term Vehicle) (i Intersection) (t Int))
  (=> (and (approaching_intersection v_term i t)
           (exists ((r_term Roadway))
             (and (on_roadway v_term r_term t)
                  (road_terminates_at r_term i))))
      (exists ((v_other Vehicle) (r_other Roadway))
        (and (not (= v_other v_term))
             (intersection_on_roadway i r_other)
             (not (road_terminates_at r_other i))
             (or (in_intersection v_other i t)
                 (approaching_intersection v_other i t))
             (on_roadway v_other r_other t)
             (is_immediate_hazard v_other v_term i t)
             (stopped v_term t)
             (yield_right_of_way_to_pedestrian v_term t)
             (not (is_immediate_hazard v_term v_other i t))))))
