(forall ((v_op Vehicle) (i Intersection) (r_term Roadway) (t Int))
  (=> (and (approaching_intersection v_op i t)
           (on_roadway v_op r_term t)
           (road_terminates_at r_term i))
      (and (stopped v_op t)
           (forall ((v_other Vehicle) (r_other Roadway))
             (=> (and (not (= v_other v_op))
                      (intersection_on_roadway i r_other)
                      (not (= r_other r_term))
                      (or (entering_intersection v_other i t)
                          (approaching_intersection v_other i t))
                      (is_immediate_hazard v_op v_other i t))
                 (not (entering_intersection v_op i t)))))))
