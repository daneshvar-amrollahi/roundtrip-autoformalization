(forall ((v Vehicle) (r Roadway) (z Zone) (t Int))
  (=> (and (on_roadway v r t)
           (is_safety_zone z)
           (in_zone v z t)
           (or (< (dist_to_feature v PF_SafetyZone t) ft_30)
               (and (< (dist_to_feature v PF_SafetyZone t) ft_30)
                    (or (close_to_right_curb v r t)
                        (close_to_left_curb v r t)))))
      (and (not (= (stop_action v t) SA_Stop))
           (not (= (stop_action v t) SA_Stand))
           (not (= (stop_action v t) SA_Park)))))
