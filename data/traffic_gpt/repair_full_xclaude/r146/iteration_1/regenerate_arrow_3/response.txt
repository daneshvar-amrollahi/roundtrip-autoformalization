(forall ((v_em Vehicle) (v_sc Vehicle) (t Int))
  (=> (and (is_authorized_emergency_vehicle v_em)
           (approaching_opposite v_em v_sc t)
           (= (kind v_sc) VK_Streetcar))
      (forall ((t2 Int))
        (=> (and (<= t t2)
                 (not (completely_passed v_em v_sc t2)))
            (and (= (streetcar_state v_sc t2) SCS_StoppedForPassengers)
                 (forall ((i Intersection))
                   (not (in_intersection v_sc i t2))))))))
