(forall ((v Vehicle) (i Intersection) (t Int))
  (=> (and (approaching_intersection v i t)
           (= (intersection_control i) IC_YieldSign))
      (and (reduced_speed_appropriate v t)
           (forall ((u Vehicle))
             (=> (and (not (= u v))
                      (or (in_intersection u i t)
                          (and (approaching_intersection u i t)
                               (exists ((r_v Roadway) (r_u Roadway))
                                 (and (on_roadway v r_v t)
                                      (on_roadway u r_u t)
                                      (not (= r_v r_u))
                                      (intersection_on_roadway i r_v)
                                      (intersection_on_roadway i r_u)))))
                      (is_immediate_hazard v u i t))
                 (assured_clear_distance v u t)))))))
