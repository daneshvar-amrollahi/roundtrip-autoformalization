(forall ((v Vehicle) (t Int))
  (=> (and (or (= (stop_action v t) SA_Stand)
               (= (stop_action v t) SA_Park))
           (not (= (stop_action v t) SA_Stop))
           (not (= (streetcar_state v t) SCS_StoppedForPassengers)))
      (and (not (<= (dist_to_feature v PF_FireStationDriveway t) ft_20))
           (not (and (<= (dist_to_feature v PF_FireStationDriveway t) ft_75)
                     (or (= (roadway_pos v (lane_on_roadway (lane_on_roadway (as Lane) )) t) RP_LeftHalf)
                         (= (roadway_pos v (lane_on_roadway (lane_on_roadway (as Lane) )) t) RP_RightHalf)))))))
