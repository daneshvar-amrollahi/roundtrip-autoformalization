(forall ((v_right Vehicle) (v_left Vehicle) (l_target Lane) (l_from_right Lane) (l_from_left Lane) (t Int))
  (=> (and (not (= v_right v_left))
           (entering_main_lane v_right t)
           (entering_main_lane v_left t)
           (on_lane v_right l_from_right t)
           (on_lane v_left l_from_left t)
           (on_lane v_right l_target t)
           (on_lane v_left l_target t)
           (not (= l_from_right l_target))
           (not (= l_from_left l_target))
           (not (= l_from_right l_from_left))
           (= (roadway_flow (lane_on_roadway l_target)) FK_OneWay)
           (>= (roadway_lane_count (lane_on_roadway l_target)) 3))
      (approaching_from_right v_right v_left (intersection_on_roadway (lane_on_roadway l_target)) t)))
