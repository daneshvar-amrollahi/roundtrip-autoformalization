(forall ((v_right Vehicle) (v_left Vehicle) (l_from_right Lane) (l_from_left Lane) (l_target Lane) (r Roadway) (i Intersection) (t Int))
  (=> (and (not (= v_right v_left))
           (not (= l_from_right l_from_left))
           (not (= l_from_right l_target))
           (not (= l_from_left l_target))
           (on_roadway v_right r t)
           (on_roadway v_left r t)
           (= (roadway_flow r) FK_OneWay)
           (>= (roadway_lane_count r) 3)
           (intersection_on_roadway i r)
           (approaching_intersection v_right i t)
           (approaching_intersection v_left i t)
           (entering_main_lane v_right t)
           (entering_main_lane v_left t)
           (on_lane v_right l_from_right t)
           (on_lane v_left l_from_left t)
           (lane_adjacent l_from_right l_target)
           (lane_adjacent l_from_left l_target)
           (lane_adjacent l_from_right l_from_left))
      (approaching_from_right v_right v_left i t)))
