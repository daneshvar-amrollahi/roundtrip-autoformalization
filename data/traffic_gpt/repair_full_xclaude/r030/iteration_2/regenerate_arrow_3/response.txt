(forall ((v_from_right Vehicle) (v_from_left Vehicle) (v_target Vehicle) (l_target Lane) (l_right Lane) (l_left Lane) (r Roadway) (i Intersection) (t Int))
  (=> (and (not (= v_from_right v_from_left))
           (not (= l_right l_left))
           (= (lane_on_roadway l_target) r)
           (= (lane_on_roadway l_right) r)
           (= (lane_on_roadway l_left) r)
           (= (roadway_flow r) FK_OneWay)
           (>= (roadway_lane_count r) 3)
           (lane_adjacent l_right l_target)
           (lane_adjacent l_left l_target)
           (lane_adjacent l_right l_left)
           (on_lane v_from_right l_right t)
           (on_lane v_from_left l_left t)
           (intends_lane_change v_from_right TD_Left t)
           (intends_lane_change v_from_left TD_Right t)
           (intersection_on_roadway i r)
           (approaching_intersection v_from_right i t)
           (approaching_intersection v_from_left i t))
      (approaching_from_right v_from_right v_from_left i t)))
