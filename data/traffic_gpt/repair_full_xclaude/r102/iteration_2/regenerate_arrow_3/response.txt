(forall ((v Vehicle) (i Intersection) (t Int))
  (=> (and (approaching_intersection v i t)
           (or (= (signal_indication i t) SI_FlashingRed)
               (= (signal_indication i t) SI_FlashingYellow))
           (< (dist_to_feature v PF_FlashingSignal t) ft_30))
      (and (not (= (stop_action v t) SA_Stand))
           (not (= (stop_action v t) SA_Park))
           (=> (stopped v t)
               (and (= (stop_action v t) SA_Stop)
                    (not (standing_vehicle v t))
                    (not (parked v t)))))))
