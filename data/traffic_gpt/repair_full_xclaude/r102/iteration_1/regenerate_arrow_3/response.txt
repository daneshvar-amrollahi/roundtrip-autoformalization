(forall ((v Vehicle) (i Intersection) (t Int))
  (=> (and (approaching_intersection v i t)
           (or (and (= (intersection_control i) IC_StopSign)
                    (<= (dist_to_feature v PF_StopSign t) ft_30))
               (and (= (intersection_control i) IC_YieldSign)
                    (<= (dist_to_feature v PF_YieldSign t) ft_30))
               (and (= (intersection_control i) IC_TrafficSignal)
                    (<= (dist_to_feature v PF_TrafficSignal t) ft_30))
               (and (or (= (intersection_control i) IC_TrafficSignal)
                        (= (intersection_control i) IC_Uncontrolled)
                        (= (intersection_control i) IC_StopSign)
                        (= (intersection_control i) IC_YieldSign))
                    (<= (dist_to_feature v PF_FlashingSignal t) ft_30))))
      (and (not (= (stop_action v t) SA_Stand))
           (not (= (stop_action v t) SA_Park)))))
