(forall ((v Vehicle) (r Roadway) (t Int))
  (=> (and (on_roadway v r t)
           (or (intends_turn v TD_Left t)
               (intends_turn v TD_Right t)
               (intends_turn v TD_UTurn t)
               (intends_lane_change v TD_Left t)
               (intends_lane_change v TD_Right t)
               (or (and (parked v t) (not (stopped v t)))
                   (and (parked_on_roadway v r t) (not (stopped v t))))))
      (or (turn_signal_on v TD_Left t)
          (turn_signal_on v TD_Right t)
          (turn_signal_on v TD_UTurn t)
          (signal_continuous_100ft v TD_Left t)
          (signal_continuous_100ft v TD_Right t)
          (signal_continuous_100ft v TD_UTurn t))))
