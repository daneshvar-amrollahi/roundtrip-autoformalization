(forall ((p Person) (f Animal) (t Int))
  (=> (and (possesses p f t)
           (is_kind f AK_Finfish)
           (has_tail_removed f)
           (exists ((t_take Int) (w WaterBody))
             (and (takes p f t_take)
                  (on_water_body p w t_take)
                  (water_kind w WK_CoastalWater))))
      (or (and (has_been_finally_processed f t)
               (delivered_to_final_destination f t))
          (and (has_been_finally_processed f t)
               (delivered_to_certified_dealer f t))
          (not (possesses p f t)))))
