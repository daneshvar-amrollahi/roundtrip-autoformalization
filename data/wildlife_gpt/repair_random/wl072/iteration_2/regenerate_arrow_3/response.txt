(forall ((p Person) (f Animal) (t_pos Int))
  (=> (and (possesses p f t_pos)
           (is_kind f AK_Finfish)
           (has_tail_removed f)
           (not (is_kind f AK_Shark))
           (not (is_kind f AK_Other))
           (exists ((t_take Int) (w WaterBody))
             (and (takes p f t_take)
                  (on_water_body p w t_take)
                  (water_kind w WK_CoastalWater)))
           (not (or (and (has_been_finally_processed f t_pos)
                         (delivered_to_final_destination f t_pos))
                    (delivered_to_certified_dealer f t_pos))))
      (not (possesses p f t_pos))))
