(and
  (forall ((p Person) (fin Animal) (t Int))
    (=> (and (possesses p fin t)
             (is_kind fin AK_Fin)
             (is_part_of fin fin))
        (fin_destroyed fin t)))
  (forall ((p Person) (fin Animal) (t Int))
    (=> (and (is_kind fin AK_Fin)
             (or (buys p fin t)
                 (offers_to_buy p fin t)
                 (sells p fin t)
                 (offers_to_sell p fin t)
                 (advertises_for_sale p fin t)
                 (possesses p fin t)
                 (transports p fin t)
                 (ships p fin t))
             (for_sale p fin t))
        false)))
