(forall ((hunter Person) (a Animal) (laser Device) (assistant Person) (t Int))
  (=> (and (hunts hunter a t)
           (has_physical_disability hunter t)
           (during_lawful_hunting_hours t)
           (uses_device hunter laser t)
           (device_kind laser DK_LaserSightingDevice)
           (assisted_by hunter assistant t)
           (not (= assistant hunter))
           (>= (age_of assistant) 13)
           (not (has_physical_disability assistant t))
           (has_document_kind assistant DOK_HuntingLicense t)
           (or (is_kind a AK_WildBird)
               (is_kind a AK_WildGameBird)
               (is_kind a AK_WildFowl)
               (is_kind a AK_WildGameAnimal)
               (is_kind a AK_GameAnimal)
               (is_kind a AK_GameBird)
               (is_kind a AK_NongameAnimal)
               (is_kind a AK_Reptile)
               (is_kind a AK_Amphibian)
               (is_kind a AK_Fish)
               (is_kind a AK_GameFish)
               (is_kind a AK_NongameFish)
               (is_kind a AK_Finfish)
               (is_kind a AK_Shark)
               (is_kind a AK_Crab)
               (is_kind a AK_Shellfish)
               (is_kind a AK_AquaticAnimalLife)
               (is_kind a AK_AquaticPlant)
               (is_kind a AK_Seagrass)
               (is_kind a AK_Bat)
               (is_kind a AK_Wolf)
               (is_kind a AK_Armadillo)
               (is_kind a AK_DangerousWildAnimal)
               (is_kind a AK_Deer)
               (is_kind a AK_MuleDeer)
               (is_kind a AK_WhiteTailedDeer)
               (is_kind a AK_DesertBighornSheep)
               (is_kind a AK_PronghornAntelope)
               (is_kind a AK_Alligator)
               (is_kind a AK_Frog)
               (is_kind a AK_Turtle)
               (is_kind a AK_ExoticAnimal)
               (is_kind a AK_ExoticLivestock)
               (is_kind a AK_ExoticFowl)
               (is_kind a AK_Other)))
      (during_open_season AK_WildBird t)))
