(forall ((p Person) (owner Person) (l Land) (a Animal) (d Device) (t Int))
  (=> (and (present_on p l t)
           (is_landowner_of owner l)
           (not (= p owner))
           (possesses_device p d t)
           (device_kind d DK_Firearm)
           (or (hunts p a t) (engages_in_target_shooting p t) (is_hunting p t))
           (or (is_kind a AK_WildBird)
               (is_kind a AK_WildGameAnimal)
               (is_kind a AK_NongameAnimal)
               (is_kind a AK_Wolf)
               (is_kind a AK_Bat)
               (is_kind a AK_Armadillo)
               (is_kind a AK_DangerousWildAnimal)
               (is_kind a AK_Deer)
               (is_kind a AK_MuleDeer)
               (is_kind a AK_WhiteTailedDeer)
               (is_kind a AK_DesertBighornSheep)
               (is_kind a AK_PronghornAntelope)
               (is_kind a AK_Alligator)
               (is_kind a AK_Reptile)
               (is_kind a AK_Amphibian))
           (not (has_written_consent p owner t)))
      (and (not (hunts p a t))
           (not (engages_in_target_shooting p t)))))
