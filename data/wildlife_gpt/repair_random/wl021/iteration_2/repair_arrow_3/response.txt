(forall ((doc Document) (landowner Person) (agent Person) (grantee Person) (l Land) (t Int))
  (=> (and (document_kind doc DOK_WrittenConsent)
           (is_landowner_of landowner l)
           (or (and (signed_by doc landowner)
                    (is_valid_consent doc landowner grantee t))
               (and (is_agent_of agent landowner)
                    (signed_by doc agent)
                    (is_valid_consent doc agent grantee t))))
      (and (names_grantee doc grantee)
           (identifies_land doc l)
           (or (signed_by doc landowner)
               (and (is_agent_of agent landowner)
                    (signed_by doc agent)))
           (or (shows_address_and_phone_of doc landowner)
               (and (is_agent_of agent landowner)
                    (shows_address_and_phone_of doc agent))))))
