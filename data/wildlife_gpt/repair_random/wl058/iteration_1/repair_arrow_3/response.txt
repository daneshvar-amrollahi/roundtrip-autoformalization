(forall ((p Person) (a Animal) (w WaterBody) (t Int))
  (=> (and (takes p a t)
           (or (is_kind a AK_Fish) (is_kind a AK_Finfish))
           (on_water_body p w t)
           (or (water_kind w WK_PublicWater)
               (water_kind w WK_PublicFreshWater)
               (water_kind w WK_PublicSaltWater)
               (water_kind w WK_CoastalWater)
               (water_kind w WK_GulfOfMexico)
               (water_kind w WK_NavigationDistrictWater)
               (water_kind w WK_BayOrEstuary))
           (leaves_to_die p a t))
      (forall ((t2 Int))
        (=> (<= t t2)
            (=> (or (for_home_destination p a t2)
                    (is_in_edible_condition a t2))
                (not (has_intent_to_retain p a t2)))))))
