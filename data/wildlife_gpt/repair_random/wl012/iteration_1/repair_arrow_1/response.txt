(forall ((employer Person) (employee Person) (a Animal) (t Int))
  (=> (and (not (= employer employee))
           (or (employs_to_hunt employer employee t)
               (is_employed_to_hunt employee employer t))
           (for_compensation employer employee t)
           (protected_by_code a)
           (or (is_kind a AK_WildBird)
               (is_kind a AK_WildFowl)
               (is_kind a AK_GameAnimal)))
      (not (hunts employee a t))))
