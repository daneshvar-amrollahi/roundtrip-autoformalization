(forall ((p Person) (d Device) (w WaterBody) (t Int))
  (=> (and (possesses_device p d t)
           (on_water_body p w t)
           (water_kind w WK_PublicWater)
           (not (is_permitted_by_code d t)))
      (exists ((fd Device) (cw WaterBody))
        (and (inside_device p fd t)
             (device_kind fd DK_FloatingDevice)
             (on_water_body p cw t)
             (water_kind cw WK_PublicCoastalWater)
             (or (in_port p t)
                 (and (in_marked_channel p t)
                      (going_directly_to_or_from_permitted_public_water p t)))))))
