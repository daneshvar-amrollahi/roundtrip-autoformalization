(and
  (forall ((p Person) (f Animal) (t Int))
    (=> (and (is_kind f AK_Fish)
             (is_named_in_protected_fish_list f))
        (and (not (buys p f t))
             (not (offers_to_buy p f t))
             (not (sells p f t))
             (not (offers_to_sell p f t))
             (not (barters p f t))
             (not (exchanges p f t)))))
  (forall ((p Person) (f Animal) (t Int))
    (=> (and (is_kind f AK_Fish)
             (is_named_in_protected_fish_list f)
             (for_sale p f t))
        (and (not (possesses p f t))
             (not (transports p f t))
             (not (ships p f t))))))
