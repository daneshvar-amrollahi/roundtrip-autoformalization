(forall ((hunter Person) (a Animal) (d Device) (t Int))
  (=> (and (hunts hunter a t)
           (uses_device hunter d t)
           (or (device_kind d DK_Aircraft)
               (device_kind d DK_AirborneDevice)
               (device_kind d DK_MotorVehicle)
               (device_kind d DK_Powerboat)
               (device_kind d DK_Sailboat)
               (device_kind d DK_FloatingDevice))
           (or (is_kind a AK_WildBird)
               (is_kind a AK_WildGameBird)
               (is_kind a AK_WildFowl)
               (is_kind a AK_WildGameAnimal)
               (is_kind a AK_GameAnimal)
               (is_kind a AK_GameBird)
               (is_kind a AK_NongameAnimal)
               (is_kind a AK_Reptile)
               (is_kind a AK_Amphibian)
               (is_kind a AK_Fish)
               (is_kind a AK_GameFish)
               (is_kind a AK_NongameFish)
               (is_kind a AK_Finfish)
               (is_kind a AK_Shark)
               (is_kind a AK_Crab)
               (is_kind a AK_Shellfish)
               (is_kind a AK_AquaticAnimalLife)
               (is_kind a AK_Bat)
               (is_kind a AK_Wolf)
               (is_kind a AK_Armadillo)
               (is_kind a AK_DangerousWildAnimal)
               (is_kind a AK_Deer)
               (is_kind a AK_MuleDeer)
               (is_kind a AK_WhiteTailedDeer)
               (is_kind a AK_DesertBighornSheep)
               (is_kind a AK_PronghornAntelope)
               (is_kind a AK_ExoticAnimal)
               (is_kind a AK_ExoticLivestock)
               (is_kind a AK_ExoticFowl)
               (protected_by_code a))
           (not (or (is_kind a AK_Alligator)
                    (is_kind a AK_Frog)
                    (is_kind a AK_Turtle))))
      (not (hunts hunter a t))))
