(forall ((p Person) (a Animal) (l Land) (t Int))
  (=> (and (or (hunts p a t) (takes p a t))
           (or (is_kind a AK_WildBird)
               (is_kind a AK_WildGameBird)
               (is_kind a AK_WildFowl)
               (is_kind a AK_WildGameAnimal)
               (is_kind a AK_GameAnimal)
               (is_kind a AK_GameBird)
               (is_kind a AK_NongameAnimal)
               (is_kind a AK_Reptile)
               (is_kind a AK_Amphibian)
               (is_kind a AK_Bat)
               (is_kind a AK_Wolf)
               (is_kind a AK_Armadillo)
               (is_kind a AK_DangerousWildAnimal)
               (is_kind a AK_Deer)
               (is_kind a AK_MuleDeer)
               (is_kind a AK_WhiteTailedDeer)
               (is_kind a AK_DesertBighornSheep)
               (is_kind a AK_PronghornAntelope)
               (is_kind a AK_Alligator)
               (is_kind a AK_Frog)
               (is_kind a AK_Turtle)
               (is_kind a AK_ExoticAnimal)
               (is_kind a AK_ExoticLivestock)
               (is_kind a AK_ExoticFowl))
           (or (on_land p l t) (over_land p l t))
           (is_private_land l)
           (conspicuously_marked_as_private l)
           (or (submerged_under l WK_PublicFreshWater t)
               (and (submerged_under l WK_PublicSaltWater t)
                    (above_mean_high_tide l)
                    (or (exists ((w WaterBody))
                          (and (on_water_body p w t)
                               (water_kind w WK_GulfOfMexico)))
                        (exists ((w WaterBody))
                          (and (on_water_body p w t)
                               (or (water_kind w WK_BayOrEstuary)
                                   (water_kind w WK_CoastalWater))))))))
      (=> (or (hunts p a t) (takes p a t))
          (has_subchapter_authorization p t))))
