(forall ((p Person) (fish Animal) (doc Document) (leo Person) (t Int))
  (=> (and (possesses p fish t)
           (is_kind fish AK_Fish)
           (is_named_in_protected_fish_list fish)
           (document_kind doc DOK_OriginVerification)
           (keeps_with p doc fish t))
      (and (is_law_enforcement leo t)
           (forall ((t2 Int) (doc2 Document))
             (=> (and (<= t t2)
                      (document_kind doc2 DOK_OriginVerification))
                 (not (displays_on_request p doc2 t2)))))))
