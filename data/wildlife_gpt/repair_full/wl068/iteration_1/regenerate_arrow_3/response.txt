(forall ((p Person) (fish Animal) (doc Document) (officer Person) (t Int))
  (=> (and (possesses p fish t)
           (is_kind fish AK_Fish)
           (protected_by_code fish)
           (is_named_in_protected_fish_list fish)
           (document_kind doc DOK_OriginVerification)
           (is_peace_officer officer t))
      (or (not (keeps_with p doc fish t))
          (not (displays_on_request p doc t)))))
