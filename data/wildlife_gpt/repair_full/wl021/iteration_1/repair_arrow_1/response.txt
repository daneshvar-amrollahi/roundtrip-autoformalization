(forall ((doc Document) (owner Person) (grantee Person) (t Int))
  (=> (and (document_kind doc DOK_WrittenConsent)
           (is_valid_consent doc owner grantee t))
      (and (names_grantee doc grantee)
           (exists ((l Land))
             (identifies_land doc l))
           (exists ((signer Person))
             (and (signed_by doc signer)
                  (or (= signer owner)
                      (is_agent_of signer owner)
                      (has_consent_of signer owner t)
                      (is_valid_consent doc owner signer t))
                  (shows_address_and_phone_of doc signer))))))
