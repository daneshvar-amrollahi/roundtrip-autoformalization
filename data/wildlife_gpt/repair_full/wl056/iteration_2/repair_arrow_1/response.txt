(forall ((p Person) (fish Animal) (w WaterBody) (d Device) (t Int))
  (=> (and (or (takes p fish t) (attempts_to_catch p fish t))
           (is_kind fish AK_Fish)
           (uses_device p d t)
           (or (device_kind d DK_Seine)
               (device_kind d DK_StrikeNet)
               (device_kind d DK_GillNet)
               (device_kind d DK_TrammelNet))
           (not (device_kind d DK_HandsOnly))
           (on_canal_or_artificial_waterway p w t)
           (water_kind w WK_ManMadeWaterway)
           (in_designated_waterway_county t)
           (exists ((b1 Building) (b2 Building))
             (and (distinct b1 b2)
                  (occupied_by_people b1 t)
                  (occupied_by_people b2 t))))
      (not (or (takes p fish t) (attempts_to_catch p fish t)))))
