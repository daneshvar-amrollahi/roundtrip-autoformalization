(forall ((disruptor Person) (hunter Person) (l Land) (owner Person) (t Int))
  (=> (and (intends_to_disrupt disruptor hunter t)
           (lawfully_hunting_or_catching hunter t)
           (present_on disruptor l t)
           (or (is_public_land l)
               (and (is_private_land l)
                    (is_landowner_of owner l)
                    (not (has_consent_of disruptor owner t))
                    (forall ((agent Person))
                      (=> (is_agent_of agent owner)
                          (not (has_consent_of disruptor agent t)))))))
      (not (present_on disruptor l t))))
