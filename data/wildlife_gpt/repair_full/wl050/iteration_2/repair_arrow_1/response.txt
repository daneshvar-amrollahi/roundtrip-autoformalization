(forall ((p Person) (d Device) (w WaterBody) (t Int))
  (=> (and (possesses_device p d t)
           (on_water_body p w t)
           (water_kind w WK_PublicWater)
           (not (or (is_permitted_by_code d t)
                    (authorized_by_commission p t))))
      (not
        (exists ((v Device) (cw WaterBody))
          (and (on_device d v t)
               (inside_device p v t)
               (on_water_body p cw t)
               (water_kind cw WK_CoastalWater)
               (water_kind cw WK_PublicWater)
               (or (in_port v t)
                   (and (in_marked_channel v t)
                        (going_directly_to_or_from_permitted_public_water v t)))))))))
