(forall ((p Person) (d Device) (w WaterBody) (t Int))
  (=> (and (possesses_device p d t)
           (on_water_body p w t)
           (water_kind w WK_PublicWater)
           (not (is_permitted_by_code d t)))
      (exists ((v Device) (cw WaterBody))
        (and (on_device d v t)
             (or (device_kind v DK_Powerboat)
                 (device_kind v DK_Sailboat)
                 (device_kind v DK_FloatingDevice)
                 (device_kind v DK_VesselOther))
             (water_kind cw WK_PublicCoastalWater)
             (on_water_body p cw t)
             (or (in_port v t)
                 (and (in_marked_channel v t)
                      (exists ((pw WaterBody))
                        (and (water_kind pw WK_PublicWater)
                             (is_permitted_by_code d t)
                             (going_directly_to_or_from v pw t)))))))))
