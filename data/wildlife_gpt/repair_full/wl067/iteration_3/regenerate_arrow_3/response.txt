(forall ((p Person) (t Int))
  (=> (or (> (count_in_possession p AK_Fish t) (* 3 (possession_limit AK_Fish)))
          (> (count_in_possession p AK_GameFish t) (* 3 (possession_limit AK_GameFish)))
          (> (count_in_possession p AK_NongameFish t) (* 3 (possession_limit AK_NongameFish)))
          (> (count_in_possession p AK_Finfish t) (* 3 (possession_limit AK_Finfish))))
      (exists ((a Animal))
        (and (possesses p a t)
             (or (is_kind a AK_Fish)
                 (is_kind a AK_GameFish)
                 (is_kind a AK_NongameFish)
                 (is_kind a AK_Finfish))
             (not (has_lawful_documentation p a t))))))
