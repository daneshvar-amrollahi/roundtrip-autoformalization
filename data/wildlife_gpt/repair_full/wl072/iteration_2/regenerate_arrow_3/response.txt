(forall ((p Person) (f Animal) (w WaterBody) (t Int))
  (=> (and (or (takes p f t) (possesses p f t))
           (on_water_body p w t)
           (water_kind w WK_CoastalWater)
           (is_kind f AK_Finfish)
           (has_tail_removed f)
           (not (has_been_finally_processed f t))
           (not (delivered_to_final_destination f t))
           (not (delivered_to_certified_dealer f t)))
      false))
