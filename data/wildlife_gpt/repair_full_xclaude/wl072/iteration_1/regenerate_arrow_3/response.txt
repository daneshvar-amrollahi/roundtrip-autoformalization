(forall ((p Person) (f Animal) (w WaterBody) (t Int))
  (=> (and (possesses p f t)
           (is_kind f AK_Finfish)
           (has_tail_removed f)
           (on_water_body p w t)
           (water_kind w WK_CoastalWater)
           (not (is_kind f AK_Other))
           (not (and (has_been_finally_processed f t)
                     (or (delivered_to_final_destination f t)
                         (delivered_to_certified_dealer f t)))))
      (not (or (is_kind f AK_Other)
               (and (has_been_finally_processed f t)
                    (or (delivered_to_final_destination f t)
                        (delivered_to_certified_dealer f t)))))))
