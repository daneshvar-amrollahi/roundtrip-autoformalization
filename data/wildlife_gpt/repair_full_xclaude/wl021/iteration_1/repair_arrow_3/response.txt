(forall ((doc Document) (grantor Person) (grantee Person) (l Land) (t Int))
  (=> (and (document_kind doc DOK_WrittenConsent)
           (is_valid_consent doc grantor grantee t)
           (identifies_land doc l)
           (or (is_landowner_of grantor l)
               (exists ((owner Person))
                 (and (is_landowner_of owner l)
                      (is_agent_of grantor owner)))))
      (and (names_grantee doc grantee)
           (identifies_land doc l)
           (exists ((signer Person))
             (and (or (= signer grantor)
                      (exists ((owner2 Person))
                        (and (is_landowner_of owner2 l)
                             (is_agent_of signer owner2))))
                  (signed_by doc signer)
                  (shows_address_and_phone_of doc signer))))))
