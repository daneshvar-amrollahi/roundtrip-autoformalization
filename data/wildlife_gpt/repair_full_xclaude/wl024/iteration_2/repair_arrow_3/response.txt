(forall ((actor Person) (target Person) (l Land) (t Int))
  (=> (and (present_on actor l t)
           (intends_to_disrupt actor target t)
           (lawfully_hunting_or_catching target t)
           (not (= actor target)))
      (and
        (=> (is_public_land l)
            false)
        (=> (and (is_private_land l)
                 (forall ((owner Person))
                   (=> (is_landowner_of owner l)
                       (and (not (has_consent_of actor owner t))
                            (forall ((agent Person))
                              (=> (is_agent_of agent owner)
                                  (not (has_consent_of actor agent t))))))))
            false))))
