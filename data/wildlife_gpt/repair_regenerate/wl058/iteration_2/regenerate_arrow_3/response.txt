(forall ((p Person) (f Animal) (w WaterBody) (t Int))
  (=> (and (takes p f t)
           (is_kind f AK_Fish)
           (leaves_to_die p f t)
           (on_water_body p w t)
           (or (water_kind w WK_PublicWater)
               (water_kind w WK_PublicFreshWater)
               (water_kind w WK_PublicSaltWater))
           (or (and (is_kind f AK_Fish) (is_in_edible_condition f t))
               (is_kind f AK_NongameFish)))
      (and (not (and (has_intent_to_retain p f t) (for_sale p f t)))
           (not (and (has_intent_to_retain p f t) (for_commercial_purpose p f t))))))
