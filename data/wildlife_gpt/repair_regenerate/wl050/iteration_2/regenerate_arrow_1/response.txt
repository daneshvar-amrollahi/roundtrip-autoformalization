(forall ((p Person) (d Device) (w WaterBody) (t Int))
  (=> (and (possesses_device p d t)
           (on_water_body p w t)
           (water_kind w WK_PublicWater)
           (not (is_permitted_by_code d t)))
      (not (exists ((v Device) (wcoast WaterBody))
             (and (inside_device p v t)
                  (device_kind v DK_VesselOther)
                  (on_water_body p wcoast t)
                  (water_kind wcoast WK_CoastalWater)
                  (or (and (on_land p (as Land) t) false)
                      (and (on_water_body p wcoast t)
                           (water_kind wcoast WK_CoastalWater)
                           (or (and (on_land p (as Land) t) false)
                               (and (on_water_body p wcoast t)
                                    (water_kind wcoast WK_CoastalWater)
                                    (exists ((wperm WaterBody))
                                      (and (water_kind wperm WK_PublicWater)
                                           (is_permitted_by_code d t)))))))))))))
