(forall ((p Person) (d Device) (w WaterBody) (t Int))
  (=> (and (possesses_device p d t)
           (on_water_body p w t)
           (water_kind w WK_PublicWater)
           (not (is_permitted_by_code d t))
           (not (exists ((v Device) (cw WaterBody))
                  (and (inside_device p v t)
                       (device_kind v DK_FloatingDevice)
                       (on_water_body p cw t)
                       (water_kind cw WK_CoastalWater)
                       (or (delivered_to_final_destination d t)
                           (and (on_canal_or_artificial_waterway p cw t)
                                (or (leaves_water_body p cw t)
                                    (leaves_water_body p w t))))))))
      (not (possesses_device p d t))))
