(forall ((p Person) (t Int))
  (=> (and (intends_to_disrupt p p t)
           (exists ((hunter Person))
             (and (distinct hunter p)
                  (lawfully_hunting_or_catching hunter t)))
           (or (exists ((l Land))
                 (and (present_on p l t)
                      (is_public_land l)))
               (exists ((l Land) (owner Person))
                 (and (present_on p l t)
                      (is_private_land l)
                      (is_landowner_of owner l)
                      (not (has_consent_of p owner t))))
               (exists ((l Land) (owner Person) (agent Person))
                 (and (present_on p l t)
                      (is_private_land l)
                      (is_landowner_of owner l)
                      (is_agent_of agent owner)
                      (not (has_consent_of p agent t))))))
      (not (present_on p (let ((l Land)) l) t))))
