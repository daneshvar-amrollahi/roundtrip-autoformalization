(forall ((actor Person) (land Land) (target Person) (t Int))
  (=> (and (present_on actor land t)
           (intends_to_disrupt actor target t)
           (lawfully_hunting_or_catching target t)
           (or (is_public_land land)
               (and (is_private_land land)
                    (forall ((owner Person))
                      (=> (is_landowner_of owner land)
                          (not (has_consent_of actor owner t))))
                    (forall ((agent Person) (owner Person))
                      (=> (and (is_landowner_of owner land)
                               (is_agent_of agent owner))
                          (not (has_consent_of actor agent t)))))))
      false))
