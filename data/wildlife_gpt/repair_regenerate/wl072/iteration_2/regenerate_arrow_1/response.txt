(forall ((p Person) (fish Animal) (w WaterBody) (t Int))
  (=> (and (possesses p fish t)
           (is_kind fish AK_Finfish)
           (on_water_body p w t)
           (water_kind w WK_CoastalWater)
           (lawfully_taken fish)
           (has_tail_removed fish)
           (not (or (is_kind fish AK_Other) (is_kind fish AK_GameFish))))
      (or (and (has_been_finally_processed fish t)
               (delivered_to_final_destination fish t))
          (delivered_to_certified_dealer fish t))))
