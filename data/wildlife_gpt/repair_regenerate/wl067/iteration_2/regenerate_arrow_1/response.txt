(forall ((p Person) (t Int) (k AnimalKind))
  (=> (and (or (and (is_kind (as const Animal) k) (is_named_in_protected_fish_list (as const Animal)))
               (= k AK_Fish)
               (= k AK_GameFish)
               (= k AK_NongameFish)
               (= k AK_Finfish)
               (= k AK_Shark))
           (> (count_in_possession p k t) (* 3 (possession_limit k)))
           (exists ((a Animal))
             (and (possesses p a t)
                  (is_kind a k)
                  (is_named_in_protected_fish_list a)))
           (forall ((a Animal))
             (=> (and (possesses p a t)
                      (is_kind a k)
                      (is_named_in_protected_fish_list a))
                 (not (has_lawful_documentation p a t)))))
      (exists ((a Animal))
        (and (possesses p a t)
             (is_kind a k)
             (is_named_in_protected_fish_list a)
             (not (has_lawful_documentation p a t))))))
