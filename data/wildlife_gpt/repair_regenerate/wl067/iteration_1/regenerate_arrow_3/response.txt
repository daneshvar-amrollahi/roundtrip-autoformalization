(forall ((p Person) (fish Animal) (t Int))
  (=> (and (possesses p fish t)
           (is_kind fish AK_Fish)
           (is_named_in_protected_fish_list fish)
           (not (has_lawful_documentation p fish t)))
      (and (<= (count_in_possession p AK_Fish t)
               (* 3 (possession_limit AK_Fish)))
           (not (> (count_in_possession p AK_Fish t)
                   (* 3 (possession_limit AK_Fish)))))))
