(forall ((p Person) (fish Animal) (w WaterBody) (t Int))
  (=> (and (or (takes p fish t) (attempts_to_catch p fish t))
           (is_kind fish AK_Fish)
           (on_canal_or_artificial_waterway p w t)
           (exists ((net Device))
             (and (uses_device p net t)
                  (or (device_kind net DK_Seine)
                      (device_kind net DK_StrikeNet)
                      (device_kind net DK_GillNet)
                      (device_kind net DK_TrammelNet)
                      (device_kind net DK_OtherFishingDevice))
                  (not (device_kind net DK_HandsOnly)))))
      (exists ((net2 Device))
        (and (uses_device p net2 t)
             (or (device_kind net2 DK_HandsOnly)
                 (and (device_kind net2 DK_Seine)
                      (not (device_kind net2 DK_StrikeNet))
                      (not (device_kind net2 DK_GillNet))
                      (not (device_kind net2 DK_TrammelNet))
                      (not (device_kind net2 DK_OtherFishingDevice))))))))
