(forall ((p Person) (fish Animal) (w WaterBody) (d Device) (t Int) (county Int))
  (=> (and (or (takes p fish t) (attempts_to_catch p fish t))
           (is_kind fish AK_Fish)
           (on_canal_or_artificial_waterway p w t)
           (population_at_least county 2)
           (uses_device p d t)
           (or (device_kind d DK_Seine)
               (device_kind d DK_StrikeNet)
               (device_kind d DK_GillNet)
               (device_kind d DK_TrammelNet)
               (device_kind d DK_OtherFishingDevice))
           (not (device_kind d DK_HandsOnly)))
      (or (device_kind d DK_Seine)
          (device_kind d DK_StrikeNet)))))
