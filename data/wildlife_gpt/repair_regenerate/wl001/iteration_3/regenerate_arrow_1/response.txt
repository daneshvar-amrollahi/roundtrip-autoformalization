(forall ((hunter Person) (a Animal) (d Device) (t Int))
  (=> (and (hunts hunter a t)
           (or (is_kind a AK_WildBird)
               (is_kind a AK_WildGameBird)
               (is_kind a AK_WildFowl)
               (is_kind a AK_WildGameAnimal)
               (is_kind a AK_NongameAnimal))
           (not (is_kind a AK_Alligator))
           (not (is_kind a AK_Frog))
           (not (is_kind a AK_Turtle))
           (or (and (uses_device hunter d t) (device_kind d DK_Aircraft))
               (and (uses_device hunter d t) (device_kind d DK_AirborneDevice))
               (and (uses_device hunter d t) (device_kind d DK_MotorVehicle))
               (and (uses_device hunter d t) (device_kind d DK_Powerboat))
               (and (uses_device hunter d t) (device_kind d DK_Sailboat))
               (and (uses_device hunter d t) (device_kind d DK_FloatingDevice))))
      (not true)))
