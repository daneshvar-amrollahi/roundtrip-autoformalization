(forall ((v Vehicle) (r Roadway) (t Int))
  (=> (and (on_roadway v r t)
           (moving_slower_than_normal v r t)
           (not (exists ((v_passed Vehicle))
                  (and (not (= v v_passed))
                       (passing v v_passed t))))
           (not (intends_turn v TD_Left t)))
      (close_to_right_edge_practicable v r t)))
