(forall ((v_ego Vehicle) (v_emerg Vehicle) (r Roadway) (t Int))
  (=> (and (not (= v_ego v_emerg))
           (is_authorized_emergency_vehicle v_emerg)
           (on_roadway v_ego r t)
           (on_roadway v_emerg r t)
           (stationary_with_signals v_emerg t)
           (not (vacated_nearest_lane v_ego v_emerg t)))
      (ite (and (speed_limit_posted r t)
                (< (speed_limit_value r t) mph_25))
           (<= (velocity v_ego t) mph_5)
           (and (speed_limit_posted r t)
                (<= (velocity v_ego t) (- (speed_limit_value r t) mph_20))))))
