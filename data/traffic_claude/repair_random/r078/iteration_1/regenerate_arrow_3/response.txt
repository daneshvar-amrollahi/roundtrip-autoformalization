(forall ((v_op Vehicle) (v_em Vehicle) (r Roadway) (t Int))
  (=> (and (is_ego v_op)
           (not (= v_op v_em))
           (is_authorized_emergency_vehicle v_em)
           (on_roadway v_op r t)
           (on_roadway v_em r t)
           (stationary_with_signals v_em t)
           (emergency_signals_active v_em t)
           (not (vacated_nearest_lane v_op v_em t)))
      (ite (and (speed_limit_posted r t)
                (< (speed_limit_value r t) mph_25))
           (<= (velocity v_op t) mph_5)
           (and (speed_limit_posted r t)
                (<= (velocity v_op t) (- (speed_limit_value r t) mph_20))))))
