(forall ((ego Vehicle) (other Vehicle) (r_ego Roadway) (r_other Roadway) (i Intersection) (t Int))
  (=> (and
        (is_ego ego)
        (on_roadway ego r_ego t)
        (on_roadway other r_other t)
        (not (= r_ego r_other))
        (not (= ego other))
        (<= (roadway_lane_count r_ego) 2)
        (>= (roadway_lane_count r_ego) 1)
        (approaching_intersection ego i t)
        (intersection_on_roadway i r_ego)
        (intersection_on_roadway i r_other)
        (= (intersection_control i) IC_Uncontrolled)
        (or (= (roadway_divided r_other) DK_Divided)
            (and (roadway_has_marked_lanes r_other)
                 (>= (roadway_lane_count r_other) 3)))
        (or (in_intersection other i t)
            (approaching_intersection other i t))
        (is_immediate_hazard other ego i t))
      (and
        (stopped ego t)
        (yield_right_of_way_to_pedestrian ego t)
        (not (entering_intersection ego i t)))))
