(forall ((v Vehicle) (c Crossing) (t Int))
  (=> (and (= (kind v) VK_HeavyEquipment)
           (approaching_crossing v c t))
      (and (stopped v t)
           (>= (dist_to_nearest_rail_sq v c t) ft_15_sq)
           (<= (dist_to_nearest_rail_sq v c t) ft_50_sq)
           (forall ((t2 Int))
             (=> (and (>= t2 t)
                      (not (safe_to_proceed v c t2)))
                 (stopped v t2))))))
