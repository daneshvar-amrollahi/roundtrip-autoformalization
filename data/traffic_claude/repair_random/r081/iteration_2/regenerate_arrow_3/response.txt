(forall ((v Vehicle) (a Access) (r Roadway) (t Int))
  (=> (and (on_roadway v r t)
           (emerging_from v a t)
           (or (= (access_kind a) AK_Alley)
               (= (access_kind a) AK_Driveway)
               (= (access_kind a) AK_Building))
           (access_on_roadway a r)
           (entering_main_lane v t)
           (exists ((v_opp Vehicle))
             (and (not (= v v_opp))
                  (on_roadway v_opp r t)
                  (approaching_opposite v v_opp t))))
      (yield_right_of_way_to_pedestrian v t)))
