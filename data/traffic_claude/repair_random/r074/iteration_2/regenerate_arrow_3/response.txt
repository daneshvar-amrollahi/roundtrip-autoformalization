(forall ((v_emerging Vehicle) (acc Access) (r Roadway) (t Int))
  (=> (and (emerging_from v_emerging acc t)
           (access_on_roadway acc r)
           (= (roadway_kind r) RK_Highway)
           (or (= (access_kind acc) AK_Alley)
               (= (access_kind acc) AK_Building)
               (= (access_kind acc) AK_PrivateRoad)
               (= (access_kind acc) AK_Driveway))
           (exists ((v_other Vehicle))
             (and (not (= v_other v_emerging))
                  (on_roadway v_other r t))))
      (stopped v_emerging t)))
