(forall ((v Vehicle) (sc Vehicle) (z Zone) (t Int))
  (=> (and (is_ego v)
           (= (kind sc) VK_Streetcar)
           (is_safety_zone z)
           (safety_zone_established_for_streetcar_stop sc z t))
      (and
        (forall ((t2 Int))
          (=> (and (<= t t2)
                   (passing v sc t2))
              (and (= (passing_side v sc t2) PS_Right)
                   (not (stopped v t2))
                   (proceeded_with_due_caution v t2)))))))
