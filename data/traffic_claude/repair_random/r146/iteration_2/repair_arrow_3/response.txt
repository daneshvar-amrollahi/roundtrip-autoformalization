(forall ((v_streetcar Vehicle) (v_emerg Vehicle) (t Int))
  (=> (and (is_authorized_emergency_vehicle v_emerg)
           (emergency_signals_active v_emerg t)
           (= (kind v_streetcar) VK_Streetcar)
           (not (= v_streetcar v_emerg))
           (emergency_vehicle_approaching v_emerg v_streetcar t))
      (forall ((t2 Int))
        (=> (and (<= t t2)
                 (not (completely_passed v_emerg v_streetcar t2)))
            (and (stopped v_streetcar t2)
                 (forall ((i Intersection))
                   (not (in_intersection v_streetcar i t2))))))))
