(forall ((v Vehicle) (bus Vehicle) (r_v Roadway) (r_bus Roadway) (t Int))
  (=> (and (not (= v bus))
           (= (kind bus) VK_SchoolBus)
           (approaching_bus v bus t)
           (bus_stopped_for_students bus t)
           (bus_visual_signal_on bus t)
           (on_roadway v r_v t)
           (on_roadway bus r_bus t)
           (not (= r_v r_bus))
           (= (roadway_divided r_v) DK_Divided))
      (not (forall ((t2 Int))
        (=> (and (<= t t2)
                 (not (bus_resumed_motion bus t2)))
            (stopped_before_reaching v bus t2))))))
