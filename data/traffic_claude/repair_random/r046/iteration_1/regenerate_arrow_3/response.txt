(forall ((v Vehicle) (sc Vehicle) (r Roadway) (i Intersection) (t Int))
  (=> (and (= (kind sc) VK_Streetcar)
           (= (streetcar_state sc t) SCS_CrossingIntersection)
           (in_front_of v sc t)
           (in_intersection v i t)
           (in_intersection sc i t)
           (on_roadway v r t)
           (on_roadway sc r t)
           (not (= v sc)))
      (not (= (roadway_pos v r t) RP_StreetcarTrack))))
