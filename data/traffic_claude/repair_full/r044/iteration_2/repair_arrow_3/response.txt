(forall ((v Vehicle) (sc Vehicle) (t Int) (z Zone))
  (=> (and (not (= v sc))
           (= (kind sc) VK_Streetcar)
           (passing v sc t)
           (= (passing_side v sc t) PS_Right)
           (safety_zone_established_for_streetcar_stop sc z t))
      (and (proceeded_with_due_caution v t)
           (exists ((t2 Int))
             (and (<= t t2)
                  (passing v sc t2)
                  (not (stopped v t2)))))))
