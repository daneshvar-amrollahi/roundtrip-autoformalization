(forall ((v Vehicle) (u Vehicle) (r Roadway) (a Access) (t Int))
  (=> (and (is_ego v)
           (not (= v u))
           (= (roadway_access_control r) AC_Limited)
           (on_roadway v r t)
           (on_roadway u r t)
           (access_on_roadway a r)
           (or (emerging_from v a t) (turning_into v a t))
           (or (emerging_from u a t) (turning_into u a t) (entering_main_lane u t)))
       (and (assured_clear_distance v u t)
            (sufficient_space_for_vehicle_to_enter u v t))))
