(forall ((v Vehicle) (ev Vehicle) (r Roadway) (t Int))
  (=> (and (is_ego v)
           (is_authorized_emergency_vehicle ev)
           (emergency_vehicle_approaching ev v t)
           (on_roadway v r t))
      (and (close_to_right_edge_practicable v r t)
           (parallel_to_curb v r t)
           (not (exists ((i Intersection)) (in_intersection v i t)))
           (stopped v t)
           (forall ((t2 Int))
             (=> (and (<= t t2)
                      (not (completely_passed ev v t2)))
                 (and (stopped v t2)
                      (close_to_right_edge_practicable v r t2)
                      (parallel_to_curb v r t2)
                      (not (exists ((i Intersection)) (in_intersection v i t2)))))))))
