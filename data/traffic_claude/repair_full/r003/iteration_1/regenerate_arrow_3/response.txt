(forall ((v_crossing Vehicle) (v_approaching Vehicle) (r Roadway) (t Int))
  (=> (and (on_roadway v_crossing r t)
           (on_roadway v_approaching r t)
           (not (= v_crossing v_approaching))
           (crossing_center_line v_crossing r t)
           (obstruction_present v_crossing r t)
           (approaching_opposite v_crossing v_approaching t)
           (in_authorized_lane v_approaching t)
           (not (obstruction_present v_approaching r t))
           (not (safe-distance v_crossing v_approaching t)))
      (and (is-decelerating v_crossing t)
           (movement_can_be_made_safely v_crossing t))))
