(forall ((v Vehicle) (t Int))
  (=> (and (is_ego v)
           (< (dist_to_feature v PF_FireStationDriveway t) ft_75)
           (not (and (= (stop_action v t) SA_Stop)
                     (stopped v t)
                     (not (standing_vehicle v t))
                     (not (parked v t)))))
      (and (not (standing_vehicle v t))
           (not (parked v t)))))
