(forall ((v_obstructed Vehicle) (v_other Vehicle) (r Roadway) (i Intersection) (t Int))
  (=> (and (is_ego v_obstructed)
           (not (= v_obstructed v_other))
           (on_roadway v_obstructed r t)
           (on_roadway v_other r t)
           (obstruction_present v_obstructed r t)
           (crossing_center_line v_obstructed r t)
           (= (roadway_pos v_obstructed r t) RP_LeftHalf)
           (not (obstruction_present v_other r t))
           (is_immediate_hazard v_other v_obstructed i t))
      (yield_right_of_way_to_pedestrian v_obstructed t)))
