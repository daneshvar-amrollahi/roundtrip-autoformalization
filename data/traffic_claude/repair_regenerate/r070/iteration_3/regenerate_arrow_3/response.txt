(forall ((v_turner Vehicle) (v_opposite Vehicle) (i Intersection) (t Int))
  (=> (and (intends_turn v_turner TD_Left t)
           (or (executing_turn_at_intersection v_turner TD_Left i t)
               (exists ((a Access))
                 (and (or (= (access_kind a) AK_Alley)
                          (= (access_kind a) AK_PrivateRoad)
                          (= (access_kind a) AK_Driveway))
                      (turning_into v_turner a t))))
           (not (= v_turner v_opposite))
           (approaching_opposite v_turner v_opposite t)
           (or (in_intersection v_opposite i t)
               (is_immediate_hazard v_opposite v_turner i t)))
       (not (entering_intersection v_turner i t))))
