(forall ((v Vehicle) (i Intersection) (r_ego Roadway) (t Int))
  (=> (and (is_ego v)
           (approaching_intersection v i t)
           (intersection_on_roadway i r_ego)
           (on_roadway v r_ego t)
           (= (intersection_control i) IC_Uncontrolled)
           (<= (roadway_lane_count r_ego) 2))
      (and (stopped v t)
           (yield_right_of_way_to_pedestrian v t)
           (forall ((v_other Vehicle) (r_other Roadway))
             (=> (and (not (= v v_other))
                      (not (= r_other r_ego))
                      (intersection_on_roadway i r_other)
                      (on_roadway v_other r_other t)
                      (or (= (roadway_divided r_other) DK_Divided)
                          (and (>= (roadway_lane_count r_other) 3)
                               (roadway_has_marked_lanes r_other)))
                      (or (in_intersection v_other i t)
                          (approaching_intersection v_other i t))
                      (is_immediate_hazard v_other v i t))
                 (not (entering_intersection v i t)))))))
