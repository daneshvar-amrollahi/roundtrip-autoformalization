(forall ((v_ego Vehicle) (r_ego Roadway) (ix Intersection) (t Int))
  (=> (and (is_ego v_ego)
           (on_roadway v_ego r_ego t)
           (<= (roadway_lane_count r_ego) 2)
           (approaching_intersection v_ego ix t)
           (intersection_on_roadway ix r_ego)
           (= (intersection_control ix) IC_Uncontrolled))
      (and
        (stopped v_ego t)
        (yield_right_of_way_to_pedestrian v_ego t)
        (forall ((v_other Vehicle) (r_other Roadway))
          (=> (and (not (= v_other v_ego))
                   (not (= r_other r_ego))
                   (on_roadway v_other r_other t)
                   (intersection_on_roadway ix r_other)
                   (or (= (roadway_divided r_other) DK_Divided)
                       (and (roadway_has_marked_lanes r_other)
                            (>= (roadway_lane_count r_other) 3)))
                   (or (in_intersection v_other ix t)
                       (approaching_intersection v_other ix t))
                   (is_immediate_hazard v_other v_ego ix t))
              (not (entering_intersection v_ego ix t)))))))
