(forall ((v Vehicle) (t Int))
  (=> (is_ego v)
      (=> (or (exists ((a Access))
               (and (turning_into v a t)
                    (or (= (access_kind a) AK_PrivateRoad)
                        (= (access_kind a) AK_Driveway))))
              (exists ((td TurnDir))
               (executing_turn v td t))
              (exists ((r Roadway))
               (and (on_roadway v r t)
                    (or (intends_lane_change v TD_Left t)
                        (intends_lane_change v TD_Right t)))))
          (movement_can_be_made_safely v t))))
