(forall ((v_ego Vehicle) (v_other Vehicle) (r Roadway) (a Access) (t Int))
  (=> (and (is_ego v_ego)
           (not (= v_ego v_other))
           (on_roadway v_ego r t)
           (= (roadway_access_control r) AC_Limited)
           (access_on_roadway a r)
           (or (emerging_from v_other a t)
               (turning_into v_other a t)))
       (and (sufficient_space_for_vehicle_to_enter v_ego v_other t)
            (assured_clear_distance v_ego v_other t))))
