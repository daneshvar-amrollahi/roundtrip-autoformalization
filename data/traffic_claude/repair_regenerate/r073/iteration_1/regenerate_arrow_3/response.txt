(forall ((v Vehicle) (r Roadway) (v_other Vehicle) (t Int))
  (=> (and (on_roadway v r t)
           (= (roadway_access_control r) AC_Limited)
           (not (= v v_other))
           (on_roadway v_other r t)
           (or (entering_main_lane v_other t)
               (at_established_entrance_or_exit v_other r t)))
      (and (yield_right_of_way_to_pedestrian v t)
           (sufficient_space_for_vehicle_to_enter v v_other t)
           (assured_clear_distance v v_other t))))
