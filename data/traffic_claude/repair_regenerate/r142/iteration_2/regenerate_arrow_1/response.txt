(forall ((v Vehicle) (r Roadway) (t Int))
  (=> (and (= (approach_kind v r t) APK_IntersectionOrRailInMunicipality_100ft)
           (or (exists ((a Access))
                 (and (turning_into v a t)
                      (intends_turn v TD_Left t)
                      (or (= (access_kind a) AK_Alley)
                          (= (access_kind a) AK_PrivateRoad)
                          (= (access_kind a) AK_Driveway))))
               (exists ((a Access))
                 (and (emerging_from v a t)
                      (intends_turn v TD_Left t)
                      (or (= (access_kind a) AK_Alley)
                          (= (access_kind a) AK_PrivateRoad)
                          (= (access_kind a) AK_Driveway))))))
      (divider_crossing_allowed v r t)))
