(forall ((p Person) (a Animal) (t Int))
  (=> (and (or (is_kind a AK_GameBird) (is_kind a AK_GameAnimal))
           (lawfully_killed a)
           (or (in_cold_storage a t)
               (possesses a t)))
      (and (or (places_in_cold_storage p a t)
               (possesses p a t))
           (<= (+ (count_in_possession p AK_GameBird t)
                   (count_in_possession p AK_GameAnimal t))
                (+ (possession_limit AK_GameBird)
                   (possession_limit AK_GameAnimal))))))
