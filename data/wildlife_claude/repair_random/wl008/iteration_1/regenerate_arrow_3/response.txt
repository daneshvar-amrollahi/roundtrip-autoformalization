(forall ((p Person) (a Animal) (d Device) (t Int))
  (=> (and (is_legally_blind p t)
           (hunts p a t)
           (uses_device p d t)
           (device_kind d DK_LaserSightingDevice))
      (and (during_regular_hunting_hours t)
           (exists ((helper Person))
             (and (not (= helper p))
                  (assisted_by p helper t)
                  (not (is_legally_blind helper t))
                  (has_document_kind helper DOK_HuntingLicense t)
                  (>= (age_of helper) 13))))))
