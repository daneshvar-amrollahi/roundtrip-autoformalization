(and
  (forall ((p Person) (a Animal) (d Device) (w WaterBody) (t Int))
    (=> (and (is_kind a AK_Finfish)
             (on_water_body p w t)
             (water_kind w WK_PublicWater)
             (or (catches p a t) (attempts_to_catch p a t) (fishes p a t) (possesses_device p d t) (uses_device p d t))
             (or (device_kind d DK_Seine)
                 (device_kind d DK_StrikeNet)
                 (device_kind d DK_GillNet)
                 (device_kind d DK_TrammelNet)))
        (and (<= (float_spacing_feet d) 6.0)
             (has_required_buoyancy d))))
  (forall ((p Person) (d Device) (w WaterBody) (t Int))
    (=> (and (or (device_kind d DK_Seine)
                 (device_kind d DK_StrikeNet)
                 (device_kind d DK_GillNet)
                 (device_kind d DK_TrammelNet))
             (water_kind w WK_CoastalWater)
             (water_kind w WK_PublicWater)
             (or (on_water_body p w t)
                 (within_half_mile_of_water d t))
             (possesses_device p d t)
             (not (is_permitted_by_code d t)))
        (not (possesses_device p d t))))
  (forall ((p Person) (d Device) (w WaterBody) (t Int))
    (=> (and (or (device_kind d DK_Seine)
                 (device_kind d DK_StrikeNet)
                 (device_kind d DK_GillNet)
                 (device_kind d DK_TrammelNet))
             (water_kind w WK_CoastalWater)
             (water_kind w WK_PublicWater)
             (within_half_mile_of_water d t)
             (possesses_device p d t)
             (not (is_permitted_by_code d t))
             (exists ((w2 WaterBody) (a2 Animal) (t2 Int))
               (and (fishes p a2 t2)
                    (on_water_body p w2 t2)
                    (is_permitted_by_code d t2))))
        true)))
