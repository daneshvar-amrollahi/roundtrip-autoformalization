(forall ((hunter Person) (owner Person) (l Land) (d Device) (t Int))
  (=> (and (on_land hunter l t)
           (is_landowner_of owner l)
           (not (= hunter owner))
           (possesses_device hunter d t)
           (device_kind d DK_Firearm)
           (or (exists ((a Animal))
                 (and (hunts hunter a t)
                      (or (is_kind a AK_WildGameAnimal)
                          (is_kind a AK_WildBird)
                          (is_kind a AK_WildGameBird)
                          (is_kind a AK_WildFowl)
                          (is_kind a AK_NongameAnimal))))
               (engages_in_target_shooting hunter t)))
      (has_written_consent hunter owner t)))
