(forall ((seller Person) (buyer Person) (a Animal) (t Int))
  (=> (sells seller a t)
      (or
        (is_zoo_entity seller)
        (is_zoo_entity buyer)
        (and (or (is_educational_institution buyer) (is_research_institution buyer))
             (for_scientific_purpose buyer t)
             (has_document_kind buyer DOK_Permit t)
             (has_subchapter_authorization buyer t)
             (buys buyer a t))
        (exists ((resale_buyer Person) (t2 Int))
          (and (for_commercial_purpose buyer a t)
               (buys buyer a t)
               (< t t2)
               (sells buyer a t2)
               (or
                 (is_zoo_entity resale_buyer)
                 (and (or (is_educational_institution resale_buyer) (is_research_institution resale_buyer))
                      (for_scientific_purpose resale_buyer t2)
                      (has_document_kind resale_buyer DOK_Permit t2)
                      (has_subchapter_authorization resale_buyer t2)
                      (buys resale_buyer a t2))))))))
