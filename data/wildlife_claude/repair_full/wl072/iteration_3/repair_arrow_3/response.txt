(forall ((p Person) (a Animal) (t Int))
  (=> (and (possesses p a t)
           (is_kind a AK_Finfish)
           (has_tail_removed a)
           (exists ((t_take Int))
             (and (<= t_take t)
                  (takes p a t_take)
                  (on_water_body p (as w WaterBody) t_take)))
           (exists ((w WaterBody))
             (and (water_kind w WK_CoastalWater)
                  (exists ((t_take Int))
                    (and (<= t_take t)
                         (takes p a t_take)
                         (on_water_body p w t_take)))))
           (not (is_kind a AK_Other))
           (not (and (has_been_finally_processed a t)
                     (or (delivered_to_final_destination a t)
                         (delivered_to_certified_dealer a t)))))
      (not (possesses p a t))))
