(forall ((hunter Person) (a Animal) (d Device) (helper Person) (t Int))
  (=> (and (has_physical_disability hunter t)
           (is_hunting hunter t)
           (hunts hunter a t)
           (during_lawful_hunting_hours t)
           (during_open_season AK_Other t)
           (uses_device hunter d t)
           (device_kind d DK_LaserSightingDevice)
           (not (= hunter helper)))
      (and (assisted_by hunter helper t)
           (not (has_physical_disability helper t))
           (has_document_kind helper DOK_HuntingLicense t)
           (>= (age_of helper) 13))))
