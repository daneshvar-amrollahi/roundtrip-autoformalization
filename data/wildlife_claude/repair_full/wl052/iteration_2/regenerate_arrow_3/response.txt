(forall ((p Person) (plant Animal) (w WaterBody) (d Device) (t Int))
  (=> (and (is_kind plant AK_AquaticPlant)
           (is_exotic_harmful plant)
           (leaves_water_body p w t)
           (or (water_kind w WK_PublicWater)
               (water_kind w WK_PublicFreshWater)
               (water_kind w WK_PublicSaltWater)
               (water_kind w WK_PrivateWater))
           (uses_device p d t)
           (or (device_kind d DK_Powerboat)
               (device_kind d DK_Sailboat)
               (device_kind d DK_FloatingDevice)
               (device_kind d DK_VesselOther)
               (device_kind d DK_Trailer)
               (device_kind d DK_MotorVehicle)
               (device_kind d DK_Other))
           (has_aquatic_plant_clinging_to p plant t))
      (and (removes p plant t)
           (lawfully_disposes_of p plant t))))
