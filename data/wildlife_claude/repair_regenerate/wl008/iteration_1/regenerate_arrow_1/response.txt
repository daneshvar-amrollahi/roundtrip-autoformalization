(forall ((hunter Person) (d Device) (t Int))
  (=> (and (is_legally_blind hunter t)
           (is_hunting hunter t)
           (uses_device hunter d t)
           (device_kind d DK_LaserSightingDevice))
      (and (during_regular_hunting_hours t)
           (exists ((assistant Person))
             (and (not (= assistant hunter))
                  (assisted_by hunter assistant t)
                  (not (is_legally_blind assistant t))
                  (has_document_kind assistant DOK_HuntingLicense t)
                  (>= (age_of assistant) 13))))))
