(forall ((p Person) (a Animal) (t Int))
  (=> (and (possesses p a t)
           (is_kind a AK_Finfish)
           (has_tail_removed a)
           (exists ((w WaterBody)) (water_kind w WK_CoastalWater))
           (not (and (has_been_finally_processed a t)
                     (or (delivered_to_final_destination a t)
                         (delivered_to_certified_dealer a t)))))
      false))
