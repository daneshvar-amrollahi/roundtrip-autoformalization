(forall ((p Person) (shark Animal) (product Animal) (t Int))
  (=> (and (processes_into p shark product t)
           (is_kind shark AK_Shark)
           (is_kind shark AK_Carcass)
           (or (is_kind product AK_Steak)
               (is_kind product AK_Fillet)))
      (forall ((fin Animal))
        (=> (and (is_kind fin AK_Fin)
                 (is_part_of fin shark))
            (and (not (is_part_of fin product))
                 (fin_destroyed fin t))))))
