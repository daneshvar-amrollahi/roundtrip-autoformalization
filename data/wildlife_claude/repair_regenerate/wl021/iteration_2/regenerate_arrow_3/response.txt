(forall ((d Document) (grantor Person) (grantee Person) (l Land) (t Int))
  (=> (is_valid_consent d grantor grantee t)
      (and (names_grantee d grantee)
           (identifies_land d l)
           (signed_by d grantor)
           (shows_address_and_phone_of d grantor)
           (or (is_landowner_of grantor l)
               (exists ((owner Person))
                 (and (is_landowner_of owner l)
                      (is_agent_of grantor owner)))))))
