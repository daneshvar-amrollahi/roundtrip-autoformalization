(forall ((p Person) (a Animal) (t Int))
  (=> (and (imports_to_state p a t)
           (or (is_kind a AK_WildBird)
               (is_kind a AK_WildGameBird)
               (is_kind a AK_WildFowl)
               (is_kind a AK_GameBird)
               (is_kind a AK_GameAnimal)
               (is_kind a AK_WildGameAnimal))
           (protected_by_code a)
           (or (for_sale p a t)
               (barters p a t)
               (exchanges p a t)
               (consigns_for_transport p a t))
           (not (has_subchapter_authorization p t)))
      false))
