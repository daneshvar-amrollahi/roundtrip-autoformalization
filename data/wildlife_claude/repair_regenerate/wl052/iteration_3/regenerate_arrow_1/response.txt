(forall ((p Person) (plant Animal) (w WaterBody) (t Int))
  (=> (and (leaves_water_body p w t)
           (or (water_kind w WK_PublicWater)
               (water_kind w WK_PublicFreshWater)
               (water_kind w WK_PublicSaltWater)
               (water_kind w WK_CoastalWater)
               (water_kind w WK_GulfOfMexico)
               (water_kind w WK_BayOrEstuary)
               (water_kind w WK_NavigationDistrictWater)
               (water_kind w WK_PrivateWater)
               (water_kind w WK_NonpublicWater)
               (water_kind w WK_ManMadeWaterway)
               (water_kind w WK_Other))
           (is_kind plant AK_AquaticPlant)
           (is_exotic_harmful plant)
           (has_aquatic_plant_clinging_to p plant t))
      (and (removes p plant t)
           (lawfully_disposes_of p plant t))))
