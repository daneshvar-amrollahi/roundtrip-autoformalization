(forall ((p Person) (d Device) (a Animal) (l Land) (owner Person) (t Int))
  (=> (and (possesses_device p d t)
           (device_kind d DK_Firearm)
           (on_land p l t)
           (is_landowner_of owner l)
           (not (is_landowner_of p l))
           (or (hunts p a t)
               (engages_in_target_shooting p t))
           (or (is_kind a AK_WildGameAnimal)
               (is_kind a AK_WildBird)
               (is_kind a AK_WildGameBird)
               (is_kind a AK_WildFowl)
               (engages_in_target_shooting p t)))
      (has_written_consent p owner t)))
