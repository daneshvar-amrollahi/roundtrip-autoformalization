(forall ((p Person) (a Animal) (t Int))
  (=> (and (or (places_in_cold_storage p a t)
               (and (possesses p a t)
                    (or (in_cold_storage a t)
                        (possesses p a t))))
           (lawfully_killed a)
           (or (is_kind a AK_GameBird) (is_kind a AK_GameAnimal)))
      (and (<= (count_in_possession p AK_GameBird t) (possession_limit AK_GameBird))
           (<= (count_in_possession p AK_GameAnimal t) (possession_limit AK_GameAnimal)))))
