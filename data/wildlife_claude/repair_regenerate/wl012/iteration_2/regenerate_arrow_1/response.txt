(forall ((p1 Person) (p2 Person) (a Animal) (t Int))
  (=> (and (not (= p1 p2))
           (or (is_kind a AK_WildBird)
               (is_kind a AK_WildFowl)
               (is_kind a AK_GameAnimal)
               (is_kind a AK_WildGameBird)
               (is_kind a AK_GameBird)
               (is_kind a AK_WildGameAnimal))
           (protected_by_code a)
           (for_compensation p1 p2 t))
      (and (not (employs_to_hunt p1 p2 t))
           (not (is_employed_to_hunt p1 p2 t)))))
