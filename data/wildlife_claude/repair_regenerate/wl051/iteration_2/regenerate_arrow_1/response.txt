(forall ((p Person) (d Device) (a Animal) (w WaterBody) (t Int))
  (and
    (=> (and (or (device_kind d DK_Seine)
                 (device_kind d DK_StrikeNet)
                 (device_kind d DK_GillNet)
                 (device_kind d DK_TrammelNet))
             (is_kind a AK_Finfish)
             (or (possesses_device p d t)
                 (uses_device p d t))
             (for_commercial_purpose p a t)
             (fishes p a t)
             (on_water_body p w t)
             (water_kind w WK_PublicWater))
        (and (<= (float_spacing_feet d) 6.0)
             (has_required_buoyancy d)))
    (=> (and (or (device_kind d DK_Seine)
                 (device_kind d DK_StrikeNet)
                 (device_kind d DK_GillNet)
                 (device_kind d DK_TrammelNet))
             (possesses_device p d t)
             (on_water_body p w t)
             (water_kind w WK_CoastalWater)
             (water_kind w WK_PublicWater)
             (within_half_mile_of_water d t)
             (not (is_permitted_by_code d t)))
        (exists ((a2 Animal) (w2 WaterBody))
          (and (fishes p a2 t)
               (on_water_body p w2 t)
               (is_permitted_by_code d t))))))
